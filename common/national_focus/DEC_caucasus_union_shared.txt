shared_focus = {
	id = DEC_CAU_chechnya_uprising
	icon = generic_fascist_uprising
	x = 10
	y = 0
	cost = 5
	allow_branch = {
		DEC_can_access_cau_uprising_branch = yes
		OR = {
			has_country_flag = DEC_CAU_branch_unlocked
			has_war_with = SOV
			DEC_has_anti_rf_focus_done = yes
			has_global_flag = DEC_TUR_block_complete
		}
	}
	search_filters = { FOCUS_FILTER_WAR_SUPPORT FOCUS_FILTER_POLITICAL }
	# Already free Chechnya (or you ARE Chechnya) → skip as completed
	bypass = {
		OR = {
			tag = CHE
			AND = { DEC_CHE_already_independent = yes NOT = { has_country_flag = DEC_uprising_refocus_CHE } }
		}
	}
	available = {
		DEC_can_fund_uprising_rifles = yes
		OR = {
			tag = CHE
			has_war_with = SOV
			SOV = { owns_state = 672 }
			CHE = { is_subject_of = SOV }
			NOT = { country_exists = CHE }
		}
	}
	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus DEC_CAU_chechnya_uprising"
		if = {
			limit = { NOT = { DEC_CHE_already_independent = yes } }
			DEC_uprising_CHE = yes
		}
		set_global_flag = DEC_CAU_any_uprising_done
		set_country_flag = DEC_CAU_branch_unlocked
		custom_effect_tooltip = DEC_uprising_independent_tt
	}
	ai_will_do = {
		base = 25
		modifier = {
			factor = 4
			has_war_with = SOV
		}
	}
}

# Caucasian Union + unlock Volga: Caucasus nations only
shared_focus = {
	id = DEC_CAU_create_union
	icon = infrastructure
	x = 0
	y = 2
	relative_position_id = DEC_CAU_dagestan_uprising
	cost = 8
	allow_branch = {
		DEC_is_caucasus_nation = yes
	}
	prerequisite = {
		focus = DEC_CAU_chechnya_uprising
		focus = DEC_CAU_dagestan_uprising
		focus = DEC_CAU_ingush_uprising
		focus = DEC_CAU_alans_uprising
		focus = DEC_CAU_circassia_uprising
	}
	search_filters = { FOCUS_FILTER_DIPLOMACY FOCUS_FILTER_POLITICAL }
	available = {
		DEC_is_caucasus_nation = yes
		is_faction_leader = no
		OR = {
			has_completed_focus = DEC_CAU_chechnya_uprising
			has_completed_focus = DEC_CAU_dagestan_uprising
			has_completed_focus = DEC_CAU_ingush_uprising
			has_completed_focus = DEC_CAU_alans_uprising
			has_completed_focus = DEC_CAU_circassia_uprising
			has_global_flag = DEC_CAU_any_uprising_done
		}
	}
	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus DEC_CAU_create_union"
		DEC_create_caucasian_union = yes
		set_global_flag = DEC_CAU_union_formed
		set_country_flag = DEC_formed_CAU
		set_country_flag = DEC_VOL_branch_unlocked
		custom_effect_tooltip = DEC_CAU_create_union_tt
		custom_effect_tooltip = DEC_CAU_unlocks_volga_tt
	}
	ai_will_do = { base = 40 }
}
