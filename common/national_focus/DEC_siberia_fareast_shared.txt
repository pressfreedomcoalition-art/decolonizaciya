	id = DEC_SIB_central_uprising
	icon = generic_fascist_uprising
	x = 100
	y = 0
	cost = 5
	allow_branch = {
		DEC_can_access_siberia_branch = yes
	}
	search_filters = { FOCUS_FILTER_WAR_SUPPORT FOCUS_FILTER_POLITICAL }
	bypass = {
		OR = {
			tag = CSB
			AND = { DEC_CSB_already_independent = yes NOT = { has_country_flag = DEC_uprising_refocus_CSB } }
		}
	}
	available = {
		DEC_can_fund_uprising_rifles = yes
		DEC_sep_siberia_ready = yes
		OR = {
			tag = CSB
			has_war_with = SOV
			SOV = { owns_state = 682 }
			NOT = { country_exists = CSB }
			DEC_CSB_already_independent = yes
			has_global_flag = DEC_MON_block_complete
			has_country_flag = DEC_SIB_branch_unlocked
		}
	}
	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus DEC_SIB_central_uprising"
		if = {
			limit = { NOT = { DEC_CSB_already_independent = yes } }
			DEC_uprising_CSB = yes
		}
		set_country_flag = DEC_SIB_branch_unlocked
		custom_effect_tooltip = DEC_uprising_independent_tt
		DEC_try_finish_siberia_block = yes
	}
	ai_will_do = { base = 14 }
}
