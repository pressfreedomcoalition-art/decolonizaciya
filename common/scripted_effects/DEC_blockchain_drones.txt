# Per-nation blockchain launch + drone armies
############################################

DEC_launch_blockchain_CHE = {
	if = {
		limit = {
			OR = {
				NOT = { country_exists = CHE }
				CHE = { check_variable = { num_owned_states < 1 } }
			}
		}
		DEC_find_free_blockchain_enclave = yes
		if = {
			limit = { check_variable = { global.DEC_bc_enclave_state > 0 } }
			var:global.DEC_bc_enclave_state = {
				add_core_of = CHE
				set_state_flag = DEC_blockchain_occupied
			}
			if = {
				limit = { NOT = { country_exists = CHE } }
				UKR = { release = CHE }
			}
			else = {
				CHE = { transfer_state = var:global.DEC_bc_enclave_state }
			}
			CHE = {
				set_country_flag = DEC_blockchain_nation
				set_country_flag = DEC_rebel_nation
				set_variable = { DEC_blockchain_state = global.DEC_bc_enclave_state }
				DEC_update_coalition = yes
				DEC_atc_on_new_rebel = yes
			}
			set_global_flag = DEC_blockchain_launched_CHE
			custom_effect_tooltip = DEC_blockchain_launch_tt
		}
		else = {
			custom_effect_tooltip = DEC_blockchain_no_enclave_tt
		}
	}
	else = {
		set_global_flag = DEC_blockchain_launched_CHE
		custom_effect_tooltip = DEC_blockchain_already_landed_tt
	}
}

DEC_drone_army_CHE = {
	set_country_flag = DEC_uprising_refocus_CHE
	uncomplete_national_focus = {
		focus = DEC_CAU_chechnya_uprising
		uncomplete_children = no
	}
	mark_focus_tree_layout_dirty = yes
	if = {
		limit = { NOT = { country_exists = CHE } }
		DEC_launch_blockchain_CHE = yes
	}
	CHE = { DEC_spawn_drone_template = yes }
	672 = {
		create_unit = {
			division = "name = \"Drone Swarm\" division_template = \"Drone Swarm\" start_experience_factor = 0.15 start_equipment_factor = 0.9"
			owner = CHE
			count = 7
		}
	}
	if = {
		limit = {
			672 = { NOT = { is_owned_by = CHE } }
		}
		672 = {
			owner = {
				CHE = {
					declare_war_on = {
						target = PREV
						type = annex_everything
					}
				}
			}
		}
	}
	if = {
		limit = { ROOT = { has_war_with = SOV } country_exists = CHE }
		CHE = {
			add_to_war = { targeted_alliance = ROOT enemy = SOV hostility_reason = asked_to_join }
		}
	}
	custom_effect_tooltip = DEC_drone_army_tt
	custom_effect_tooltip = DEC_uprising_refocus_tt
}
