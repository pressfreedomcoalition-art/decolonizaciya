DEC_CAL_add_rad_islam_amt = {
	if = {
		limit = { NOT = { has_variable = party_pop_array^11 } }
		set_variable = { party_pop_array^11 = 0 }
	}
	add_to_variable = { party_pop_array^11 = DEC_cal_rad_amt }
	clamp_variable = { var = party_pop_array^11 min = 0 max = 1 }
	add_popularity = { ideology = fascism popularity = DEC_cal_rad_amt }
	recalculate_party = yes
}

DEC_create_world_caliphate = {
	DEC_mark_named_alliance_member = yes
	if = { limit = { is_in_faction = yes } leave_faction = yes }
	set_rule = { can_create_factions = yes }
	create_faction_from_template = faction_template_dec_world_caliphate
	set_global_flag = DEC_CAL_world_caliphate_formed
}

DEC_invite_caliphate_members = {
	every_other_country = {
		limit = {
			DEC_is_caliphate_invitee = yes
			NOT = { is_in_faction_with = ROOT }
		}
		set_country_flag = DEC_joined_named_faction
		if = { limit = { is_in_faction = yes } leave_faction = yes }
		ROOT = { add_to_faction = PREV }
	}
}

DEC_CAL_bayah_start = {
	set_country_flag = DEC_CAL_bayah_caller
	save_global_event_target_as = DEC_caliph_overlord
	every_other_country = {
		limit = {
			is_in_faction_with = ROOT
			is_subject = no
			NOT = { tag = ROOT }
		}
		country_event = { id = dec_cal.1 hours = 8 }
	}
}

DEC_CAL_unite_caliphate = {
	every_other_country = {
		limit = {
			OR = {
				is_subject_of = ROOT
				has_country_flag = DEC_CAL_bayah_accepted
			}
		}
		every_owned_state = { add_core_of = ROOT }
		ROOT = { annex_country = { target = PREV transfer_troops = yes } }
	}
	set_cosmetic_tag = KHF
	set_global_flag = DEC_CAL_united
	custom_effect_tooltip = DEC_CAL_unite_caliphate_tt
}

DEC_CAL_uprising_NCY = {
	log = "DEC_CAL: jihadist uprising NCY"
	if = {
		limit = {
			country_exists = NCY
			NCY = { DEC_has_radical_islam = yes }
		}
		custom_effect_tooltip = DEC_CAL_uprising_already_tt
	}
	else_if = {
		limit = {
			country_exists = NCY
			NCY = { is_subject = no }
		}
		NCY = {
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			set_variable = { party_pop_array^11 = 0.45 }
			add_popularity = { ideology = fascism popularity = 0.45 }
			recalculate_party = yes
			start_civil_war = {
				ideology = fascism
				size = 0.45
			}
		}
		random_country = {
			limit = {
				original_tag = NCY
				has_government = fascism
				has_war_with = NCY
			}
			jihadist_conversion_setup = yes
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = PREV }
				set_country_flag = DEC_joined_named_faction
			}
		}
	}
	else_if = {
		limit = { country_exists = NCY }
		# subject — free and convert
		NCY = {
			OVERLORD = { set_temp_variable = { DEC_cal_overlord = THIS.id } }
			jihadist_conversion_setup = yes
			if = { limit = { is_subject = yes } OVERLORD = { set_autonomy = { target = PREV autonomy_state = autonomy_free } } }
		}
		if = {
			limit = { ROOT = { is_in_faction = yes } }
			ROOT = { add_to_faction = NCY }
		}
		NCY = {
			if = {
				limit = { has_variable = DEC_cal_overlord }
				declare_war_on = { target = var:DEC_cal_overlord type = annex_everything }
			}
		}
	}
	else = {
		# country missing — release cores from any owner
		random_state = {
			limit = { is_core_of = NCY }
			OWNER = {
				release = NCY
			}
		}
		if = {
			limit = { country_exists = NCY }
			NCY = {
				jihadist_conversion_setup = yes
				every_owned_state = {
					limit = { is_core_of = NCY }
					# keep
				}
			}
			random_state = {
				limit = { is_core_of = NCY NOT = { is_owned_by = NCY } }
				OWNER = { set_temp_variable = { DEC_cal_occ = THIS.id } }
			}
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = NCY }
			}
			NCY = {
				if = {
					limit = { has_variable = DEC_cal_occ }
					declare_war_on = { target = var:DEC_cal_occ type = annex_everything }
				}
				else_if = {
					limit = { any_neighbor_country = { always = yes } }
					random_neighbor_country = {
						PREV = { declare_war_on = { target = PREV type = annex_everything } }
					}
				}
			}
		}
	}
