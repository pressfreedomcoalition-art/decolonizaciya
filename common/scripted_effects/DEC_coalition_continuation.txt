DEC_create_atc = {
	set_global_flag = DEC_atc_created
	# NAP among all rebel nations
	every_country = {
		limit = {
			has_country_flag = DEC_rebel_nation
			exists = yes
			is_subject = no
		}
		every_other_country = {
			limit = {
				has_country_flag = DEC_rebel_nation
				exists = yes
				is_subject = no
				NOT = { has_non_aggression_pact_with = PREV }
			}
			diplomatic_relation = {
				country = PREV
				relation = non_aggression_pact
				active = yes
			}
		}
	}
	# NATO memorandum invitations
	every_country = {
		limit = {
			is_in_array = { global.nato_members = THIS }
			NOT = { has_country_flag = DEC_atc_memo_asked }
			NOT = { has_war_with = UKR }
		}
		set_country_flag = DEC_atc_memo_asked
		country_event = { id = dec.30 hours = 8 }
	}
	custom_effect_tooltip = DEC_COA_atc_tt
}

# When a new rebel appears after ATC — give NAP with rebels + memo NATO
DEC_atc_on_new_rebel = {
	if = {
		limit = { has_global_flag = DEC_atc_created }
		# THIS = new rebel
		every_other_country = {
			limit = {
				OR = {
					has_country_flag = DEC_rebel_nation
					has_country_flag = DEC_atc_memo_accepted
				}
				exists = yes
				is_subject = no
				NOT = { has_non_aggression_pact_with = PREV }
			}
			diplomatic_relation = {
				country = PREV
				relation = non_aggression_pact
				active = yes
			}
		}
	}
}

DEC_find_free_blockchain_enclave = {
	# Sets global.DEC_bc_enclave_state to free UKR-owned enclave, or 0
	# Uses any_state so missing enclave state IDs (when blockchain is disabled) do not error
	set_variable = { global.DEC_bc_enclave_state = 0 }
	random_state = {
		limit = {
			is_owned_by = UKR
			has_state_flag = DEC_blockchain_enclave
			NOT = { has_state_flag = DEC_blockchain_occupied }
		}
		set_variable = { global.DEC_bc_enclave_state = THIS.id }
	}
}

DEC_return_blockchain_enclave = {
	# THIS = nation that had blockchain capital
	if = {
		limit = { has_country_flag = DEC_blockchain_nation check_variable = { DEC_blockchain_state > 0 } }
		var:DEC_blockchain_state = {
			clr_state_flag = DEC_blockchain_occupied
			UKR = { transfer_state = PREV }
			remove_core_of = PREV
		}
		clr_country_flag = DEC_blockchain_nation
		set_variable = { DEC_blockchain_state = 0 }
		log = "DEC: blockchain enclave returned to UKR from [THIS.GetTag]"
	}
}

DEC_check_blockchain_return = {
	# THIS = country — if blockchain nation owns any non-enclave state, return enclave
	if = {
		limit = {
			has_country_flag = DEC_blockchain_nation
			any_owned_state = {
				NOT = { has_state_flag = DEC_blockchain_enclave }
			}
		}
		DEC_return_blockchain_enclave = yes
	}
}

DEC_apply_historical_claims = {
	every_country = {
		limit = {
			OR = {
				has_country_flag = DEC_rebel_nation
				DEC_is_uprising_nation_tag = yes
			}
		}
		every_state = {
			limit = {
				is_core_of = PREV
				OR = {
					is_owned_by = SOV
					owner = { is_subject_of = SOV }
				}
				NOT = { is_claimed_by = PREV }
			}
			add_claim_by = PREV
		}
	}
	every_state = {
		limit = {
			OR = {
				is_owned_by = SOV
				owner = { is_subject_of = SOV }
			}
			NOT = { DEC_is_muscovy_state = yes }
			NOT = {
				any_country = {
					DEC_is_uprising_nation_tag = yes
					PREV = { is_claimed_by = PREV }
				}
			}
			NOT = {
				any_country = {
					DEC_is_uprising_nation_tag = yes
					PREV = { is_core_of = PREV }
				}
			}
		}
		DEC_assign_residual_claim = yes
	}
	custom_effect_tooltip = DEC_COA_historical_claims_tt
}

DEC_assign_residual_claim = {
	# THIS = state
	if = {
		limit = { OR = { state = 680 state = 681 state = 682 state = 685 state = 1067 state = 1069 state = 688 state = 689 } }
		add_claim_by = SIB
	}
	else_if = {
		limit = { OR = { state = 648 state = 678 state = 679 state = 683 state = 675 state = 676 state = 658 } }
		add_claim_by = URA
	}
	else_if = {
		limit = { OR = { state = 667 state = 668 state = 1118 state = 1140 } }
		add_claim_by = KUB
	}
	else_if = {
		limit = { is_on_continent = europe }
		add_claim_by = SEV
	}
	else = {
		add_claim_by = SIB
	}
}
