DEC_uprising_spawn_national_militia = {
	division_template = {
		name = "National Volunteers"
		is_locked = yes
		regiments = {
			Militia_Bat = { x = 0 y = 0 }
			Militia_Bat = { x = 0 y = 1 }
			Militia_Bat = { x = 1 y = 0 }
			Militia_Bat = { x = 1 y = 1 }
		}
		priority = 2
	}
}

DEC_uprising_TAT = {
	log = "DEC: TAT uprising vs SOV"
	661 = { add_core_of = TAT }
	if = {
		limit = { NOT = { country_exists = TAT } }
		SOV = { release = TAT }
	}
	else = {
		every_state = {
			limit = {
				is_core_of = TAT
				OR = {
					is_owned_by = SOV
					owner = { is_subject_of = SOV }
				}
			}
			TAT = { transfer_state = PREV }
		}
	}
	TAT = {
		DEC_uprising_free_from_SOV = yes
		clr_country_flag = SOV_subject_republic
		DEC_uprising_spawn_national_militia = yes
		661 = {
			create_unit = {
				division = "name = \"Tatar Brigade\" division_template = \"National Volunteers\" start_experience_factor = 0.2 start_equipment_factor = 0.75"
				owner = TAT
				prioritize_location = 6338
				count = 3
			}
		}
		declare_war_on = { target = SOV type = annex_everything }
	}
	if = {
		limit = { ROOT = { has_war_with = SOV } }
		TAT = {
			add_to_war = { targeted_alliance = ROOT enemy = SOV hostility_reason = asked_to_join }
		}
	}
	set_global_flag = DEC_TAT_independent_uprising
}
