DEC_nato_invite_finno_to_finland = {
	if = {
		limit = {
			country_exists = FIN
			NOT = { FIN = { is_in_faction = yes } }
		}
		FIN = {
			create_faction_from_template = faction_template_dec_fennoscandia
		}
	}
	every_country = {
		limit = {
			DEC_is_finno_ugric_collapse = yes
			exists = yes
			is_subject = no
			NOT = { tag = FIN }
			NOT = { is_in_faction_with = FIN }
		}
		if = {
			limit = { FIN = { is_faction_leader = yes } }
			FIN = { add_to_faction = PREV }
		}
		else_if = {
			limit = { FIN = { is_in_faction = yes } }
			FIN = {
				faction_leader = {
					add_to_faction = PREVPREV
				}
			}
		}
		country_event = { id = dec.21 hours = 2 }
	}
}

DEC_rdk_start_revolution = {
	if = {
		limit = {
			country_exists = RUS
			RUS = {
				OR = {
					has_government = democratic
					has_cosmetic_tag = RUS_RFOR
				}
				NOT = { has_civil_war = yes }
				NOT = { has_country_flag = DEC_RDK_revolution_fired }
			}
		}
		RUS = {
			set_country_flag = DEC_RDK_revolution_fired
			country_event = { id = dec.20 hours = 8 }
		}
	}
	else_if = {
		limit = {
			country_exists = SOV
			SOV = {
				has_government = democratic
				NOT = { has_civil_war = yes }
				NOT = { has_country_flag = DEC_RDK_revolution_fired }
			}
			NOT = { country_exists = RUS }
		}
		SOV = {
			set_country_flag = DEC_RDK_revolution_fired
			country_event = { id = dec.20 hours = 8 }
		}
	}
}
