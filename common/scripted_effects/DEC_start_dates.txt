DEC_map_2013 = {
	# Ensure South Sudan exists on its cores (MD 2000 often already owns; re-assert)
	if = {
		limit = { NOT = { country_exists = SSU } }
		SUD = {
			release = SSU
		}
	}
	else = {
		every_state = {
			limit = {
				is_core_of = SSU
				is_owned_by = SUD
			}
			SSU = { transfer_state = PREV }
		}
	}
}

DEC_map_2022 = {
	# Crimea to Russia
	669 = { add_core_of = SOV }
	if = {
		limit = { NOT = { 669 = { is_owned_by = SOV } } }
		SOV = { transfer_state = 669 }
	}
	# Donetsk / Luhansk cities → DPR / LPR cores
	693 = { add_core_of = DPR }
	1075 = { add_core_of = LPR }
	# Release DPR/LPR from Ukraine if not already independent
	if = {
		limit = {
			NOT = { country_exists = DPR }
			UKR = { owns_state = 693 }
		}
		UKR = { release = DPR }
	}
	else_if = {
		limit = {
			country_exists = DPR
			UKR = { owns_state = 693 }
		}
		DPR = { transfer_state = 693 }
	}
	if = {
		limit = {
			NOT = { country_exists = LPR }
			UKR = { owns_state = 1075 }
		}
		UKR = { release = LPR }
	}
	else_if = {
		limit = {
			country_exists = LPR
			UKR = { owns_state = 1075 }
		}
		LPR = { transfer_state = 1075 }
	}
}

# Quiet NATO admit for ROOT (no news events). Faction first — NATO_member cancels if not in faction.
DEC_nato_admit_quiet = {
	if = {
		limit = { exists = yes }
		USA = {
			if = {
				limit = {
					is_faction_leader = yes
					NOT = { is_in_faction_with = PREV }
				}
				add_to_faction = PREV
			}
		}
		if = {
			limit = { NOT = { has_idea = NATO_member } }
			add_ideas = NATO_member
			add_to_tech_sharing_group = NATO_Tech_Share
		}
		if = {
			limit = { NOT = { is_in_array = { global.nato_members = THIS } } }
			add_to_array = { global.nato_members = THIS }
		}
	}
}

DEC_diplomacy_2013 = {
	log = "DEC: diplomacy 2013 applied"

	# --- NATO: sync idea + faction for MD late-start array members ---
	if = { limit = { country_exists = EST } EST = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = LAT } LAT = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = LIT } LIT = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = SLO } SLO = { DEC_nato_admit_quiet = yes } } # Slovakia
	if = { limit = { country_exists = SLV } SLV = { DEC_nato_admit_quiet = yes } } # Slovenia
	if = { limit = { country_exists = ROM } ROM = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = BUL } BUL = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = ALB } ALB = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = CRO } CRO = { DEC_nato_admit_quiet = yes } }

	# --- End Second Chechen War leftover; Kadyrov as RF satellite ---
	if = {
		limit = {
			country_exists = CHE
			SOV = { has_war_with = CHE }
		}
		SOV = { white_peace = CHE }
	}
	if = {
		limit = {
			country_exists = CHE
			NOT = { CHE = { is_subject_of = SOV } }
		}
		SOV = { puppet = CHE }
	}

	# --- End MD-2000 wars that were over by Oct 2013 ---
	if = {
		limit = {
			country_exists = SRI
			country_exists = TML
			SRI = { has_war_with = TML }
		}
		SRI = { white_peace = TML }
	}
	if = {
		limit = {
			country_exists = IND
			country_exists = ACE
			IND = { has_war_with = ACE }
		}
		IND = { white_peace = ACE }
	}
	if = {
		limit = {
			country_exists = ERI
			country_exists = ETH
			ERI = { has_war_with = ETH }
		}
		ERI = { white_peace = ETH }
	}
	if = {
		limit = {
			country_exists = NEP
			country_exists = NPM
			NEP = { has_war_with = NPM }
		}
		NEP = { white_peace = NPM }
	}
	if = {
		limit = {
			country_exists = LUR
			country_exists = LIB
			LUR = { has_war_with = LIB }
		}
		LUR = { white_peace = LIB }
	}
	if = {
		limit = {
			country_exists = AFR
			country_exists = SIE
			AFR = { has_war_with = SIE }
		}
		AFR = { white_peace = SIE }
	}
	if = {
		limit = {
			country_exists = UNI
			country_exists = AGL
			UNI = { has_war_with = AGL }
		}
		UNI = { white_peace = AGL }
	}
	if = {
		limit = {
			country_exists = SUD
			country_exists = SSU
			SUD = { has_war_with = SSU }
		}
		SUD = { white_peace = SSU }
	}

	# --- Post-2008 Caucasus recognition (RF) ---
	if = {
		limit = { country_exists = ABK }
		SOV = {
			add_opinion_modifier = { target = ABK modifier = historic_friends }
			reverse_add_opinion_modifier = { target = ABK modifier = historic_friends }
			diplomatic_relation = {
				country = ABK
				relation = military_access
				active = yes
			}
			diplomatic_relation = {
				country = ABK
				relation = guarantee
				active = yes
			}
		}
	}
	if = {
		limit = { country_exists = SOO }
		SOV = {
			add_opinion_modifier = { target = SOO modifier = historic_friends }
			reverse_add_opinion_modifier = { target = SOO modifier = historic_friends }
			diplomatic_relation = {
				country = SOO
				relation = military_access
				active = yes
			}
			diplomatic_relation = {
				country = SOO
				relation = guarantee
				active = yes
			}
		}
	}
	# Drop leftover MD-2000 SOV access into Georgia after 2008 war
	if = {
		limit = {
			country_exists = GEO
			SOV = { has_military_access_to = GEO }
		}
		SOV = {
			diplomatic_relation = {
				country = GEO
				relation = military_access
				active = no
			}
		}
	}
}

DEC_diplomacy_2022 = {
	log = "DEC: diplomacy 2022 applied"

	# --- NATO 2017 / 2020 expansions (not in MD late array) ---
	if = { limit = { country_exists = MNT } MNT = { DEC_nato_admit_quiet = yes } }
	if = { limit = { country_exists = FYR } FYR = { DEC_nato_admit_quiet = yes } }

	# --- Donbas: frozen-hot war since 2014; RF recognised DPR/LPR 21.02 ---
	if = {
		limit = {
			country_exists = DPR
			country_exists = UKR
			NOT = { UKR = { has_war_with = DPR } }
		}
		UKR = {
			declare_war_on = {
				target = DPR
				type = annex_everything
			}
		}
	}
	if = {
		limit = {
			country_exists = LPR
			country_exists = UKR
			NOT = { UKR = { has_war_with = LPR } }
		}
		UKR = {
			declare_war_on = {
				target = LPR
				type = annex_everything
			}
		}
	}
	if = {
		limit = { country_exists = DPR }
		SOV = {
			add_opinion_modifier = { target = DPR modifier = DEC_decolonised_solidarity }
			add_opinion_modifier = { target = DPR modifier = historic_friends }
			diplomatic_relation = {
				country = DPR
				relation = non_aggression_pact
				active = yes
			}
			diplomatic_relation = {
				country = DPR
				relation = military_access
				active = yes
			}
			diplomatic_relation = {
				country = DPR
				relation = guarantee
				active = yes
			}
		}
		DPR = {
			diplomatic_relation = {
				country = SOV
				relation = military_access
				active = yes
			}
		}
	}
	if = {
		limit = { country_exists = LPR }
		SOV = {
			add_opinion_modifier = { target = LPR modifier = DEC_decolonised_solidarity }
			add_opinion_modifier = { target = LPR modifier = historic_friends }
			diplomatic_relation = {
				country = LPR
				relation = non_aggression_pact
				active = yes
			}
			diplomatic_relation = {
				country = LPR
				relation = military_access
				active = yes
			}
			diplomatic_relation = {
				country = LPR
				relation = guarantee
				active = yes
			}
		}
		LPR = {
			diplomatic_relation = {
				country = SOV
				relation = military_access
				active = yes
			}
		}
	}
	# Mutual Donbas coordination
	if = {
		limit = {
			country_exists = DPR
			country_exists = LPR
		}
		DPR = {
			diplomatic_relation = {
				country = LPR
				relation = military_access
				active = yes
			}
			diplomatic_relation = {
				country = LPR
				relation = non_aggression_pact
				active = yes
			}
		}
		LPR = {
			diplomatic_relation = {
				country = DPR
				relation = military_access
				active = yes
			}
		}
	}

	# --- Afghanistan: Taliban victory (Aug 2021); AFG tag keeps government ---
	if = {
		limit = {
			country_exists = TAL
			country_exists = AFG
			TAL = { has_war_with = AFG }
		}
		TAL = { white_peace = AFG }
	}
	if = {
		limit = { country_exists = TAL }
		AFG = {
			annex_country = {
				target = TAL
				transfer_troops = yes
			}
		}
	}
}
