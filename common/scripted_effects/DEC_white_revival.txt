DEC_WHT_add_ns_amt = {
	if = {
		limit = { NOT = { has_variable = party_pop_array^21 } }
		set_variable = { party_pop_array^21 = 0 }
	}
	add_to_variable = { party_pop_array^21 = DEC_wht_ns_amt }
	clamp_variable = { var = party_pop_array^21 min = 0 max = 1 }
	add_popularity = { ideology = nationalist popularity = DEC_wht_ns_amt }
	recalculate_party = yes
}

DEC_WHT_ns_conversion_setup = {
	set_variable = { party_pop_array^21 = 0.55 }
	add_popularity = { ideology = nationalist popularity = 0.55 }
	set_variable = { ruling_party = 21 }
	recalculate_party = yes
	set_politics = { ruling_party = nationalist elections_allowed = no }
	set_country_flag = DEC_rebel_nation
	if = {
		limit = { OR = { original_tag = BLT tag = BLT } }
		DEC_BLT_bdv_victory = yes
	}
}

	if = {
		limit = { country_exists = ENG }
		ENG = {
			set_temp_variable = { DEC_wht_ns_amt = ROOT.DEC_wht_ns_amt }

		}
	}
}
DEC_WHT_uprising_ENG = {
	log = "DEC_WHT: NS uprising ENG"
	if = {
		limit = {
			country_exists = ENG
			ENG = { DEC_has_ns = yes }
		}
		custom_effect_tooltip = DEC_WHT_uprising_already_tt
	}
	else_if = {
		limit = {
			country_exists = ENG
			ENG = { is_subject = no }
		}
		ENG = {
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			set_variable = { party_pop_array^21 = 0.45 }
			add_popularity = { ideology = nationalist popularity = 0.45 }
			recalculate_party = yes
			start_civil_war = {
				ideology = nationalist
				size = 0.45
			}
		}
		random_country = {
			limit = {
				original_tag = ENG
				has_government = nationalist
				has_war_with = ENG
			}
			DEC_WHT_ns_conversion_setup = yes
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = PREV }
				set_country_flag = DEC_joined_named_faction
			}
		}
	}
	else_if = {
		limit = { country_exists = ENG }
		ENG = {
			if = {
				limit = { is_subject = yes }
				OVERLORD = { save_event_target_as = DEC_wht_former_overlord }
				OVERLORD = { set_autonomy = { target = PREV autonomy_state = autonomy_free } }
			}
			DEC_WHT_ns_conversion_setup = yes
		}
		if = {
			limit = { ROOT = { is_in_faction = yes } }
			ROOT = { add_to_faction = ENG }
		}
		ENG = {
			if = {
				limit = { has_event_target = DEC_wht_former_overlord }
				declare_war_on = { target = event_target:DEC_wht_former_overlord type = annex_everything }
			}
		}
	}
	else = {
		random_state = {
			limit = { is_core_of = ENG }
			OWNER = { release = ENG }
		}
		if = {
			limit = { country_exists = ENG }
			ENG = { DEC_WHT_ns_conversion_setup = yes }
			random_state = {
				limit = { is_core_of = ENG NOT = { is_owned_by = ENG } }
				OWNER = { save_event_target_as = DEC_wht_occ }
			}
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = ENG }
			}
			ENG = {
				if = {
					limit = { has_event_target = DEC_wht_occ }
					declare_war_on = { target = event_target:DEC_wht_occ type = annex_everything }
				}
			}
		}
	}
}

	if = {
		limit = {
			country_exists = ALB
			ALB = { DEC_has_ns = yes }
		}
		custom_effect_tooltip = DEC_WHT_uprising_already_tt
	}
	else_if = {
		limit = {
			country_exists = ALB
			ALB = { is_subject = no }
		}
		ALB = {
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			set_variable = { party_pop_array^21 = 0.45 }
			add_popularity = { ideology = nationalist popularity = 0.45 }
			recalculate_party = yes
			start_civil_war = {
				ideology = nationalist
				size = 0.45
			}
		}
		random_country = {
			limit = {
				original_tag = ALB
				has_government = nationalist
				has_war_with = ALB
			}
			DEC_WHT_ns_conversion_setup = yes
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = PREV }
				set_country_flag = DEC_joined_named_faction
			}
		}
	}
	else_if = {
		limit = { country_exists = ALB }
		ALB = {
			if = {
				limit = { is_subject = yes }
				OVERLORD = { save_event_target_as = DEC_wht_former_overlord }
				OVERLORD = { set_autonomy = { target = PREV autonomy_state = autonomy_free } }
			}
			DEC_WHT_ns_conversion_setup = yes
		}
		if = {
			limit = { ROOT = { is_in_faction = yes } }
			ROOT = { add_to_faction = ALB }
		}
		ALB = {
			if = {
				limit = { has_event_target = DEC_wht_former_overlord }
				declare_war_on = { target = event_target:DEC_wht_former_overlord type = annex_everything }
			}
		}
	}
	else = {
		random_state = {
			limit = { is_core_of = ALB }
			OWNER = { release = ALB }
		}
		if = {
			limit = { country_exists = ALB }
			ALB = { DEC_WHT_ns_conversion_setup = yes }
			random_state = {
				limit = { is_core_of = ALB NOT = { is_owned_by = ALB } }
				OWNER = { save_event_target_as = DEC_wht_occ }
			}
			if = {
				limit = { ROOT = { is_in_faction = yes } }
				ROOT = { add_to_faction = ALB }
			}
			ALB = {
				if = {
					limit = { has_event_target = DEC_wht_occ }
					declare_war_on = { target = event_target:DEC_wht_occ type = annex_everything }
				}
			}
