# Exile / no real capital (decolonizers in GIE or blockchain-only)
DEC_is_exile_without_capital = {
	OR = {
		is_government_in_exile = yes
		has_country_flag = DEC_government_in_exile
		check_variable = { num_owned_states < 1 }
		AND = {
			has_country_flag = DEC_blockchain_nation
			NOT = {
				any_owned_state = {
					NOT = { has_state_flag = DEC_blockchain_enclave }
				}
			}
		}
	}
}

# Shared exile independence branch (stays open after homeland landing for aid focuses)
DEC_can_access_exile_branch = {
	NOT = {
		OR = {
			original_tag = UKR
			tag = UKR
		}
	}
	OR = {
		has_country_flag = DEC_exile_path_open
		AND = {
			OR = {
				has_country_flag = DEC_rebel_nation
				DEC_is_uprising_nation_tag = yes
			}
			DEC_is_exile_without_capital = yes
		}
	}
}

	OR = {
		AND = {
			OR = { original_tag = UKR tag = UKR }
			has_country_flag = DEC_UKR_coalition_access
		}
		AND = {
			has_country_flag = DEC_rebel_nation
			NOT = { has_country_flag = DEC_joined_named_faction }
			exists = yes
			is_subject = no
		}

}
