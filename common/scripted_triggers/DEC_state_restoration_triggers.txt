DEC_can_access_state_restoration = {
	OR = {
		DEC_is_restoration_nation = yes
		AND = {
			is_subject = no
			NOT = {
				OR = {
					original_tag = SOV
					tag = SOV
					original_tag = UKR
					tag = UKR
					original_tag = CHI
					tag = CHI
					original_tag = PER
					tag = PER
					original_tag = RUS
					tag = RUS
				}
			}
			OR = {
				has_war_with = SOV
				has_war_with = UKR
				has_war_with = CHI
				has_war_with = PER
				any_core_state = {
					OR = {
						is_owned_by = SOV
						is_owned_by = UKR
						is_owned_by = CHI
						is_owned_by = PER
						owner = { is_subject_of = SOV }
						owner = { is_subject_of = UKR }
						owner = { is_subject_of = CHI }
						owner = { is_subject_of = PER }
					}
				}
			}
		}
	}
}
