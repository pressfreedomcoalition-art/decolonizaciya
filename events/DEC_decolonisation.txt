# SOV: Volga Bulgaria declares restoration (releases VBG on Chuvashia core land)
country_event = {
	id = dec.2
	title = dec.2.t
	desc = dec.2.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = dec.2.a
		log = "[GetDateText]: [This.GetName]: dec.2.a executed"
		if = {
			limit = { NOT = { country_exists = VBG } }
			1135 = { add_core_of = VBG }
			release = VBG
			VBG = {
				set_politics = {
					ruling_party = democratic
					elections_allowed = yes
				}
				country_event = { id = dec.4 hours = 2 }
			}
		}
		ai_chance = { base = 40 }
	}

	option = {
		name = dec.2.b
		log = "[GetDateText]: [This.GetName]: dec.2.b executed"
		add_stability = -0.05
		add_political_power = -50
		1135 = {
			add_core_of = VBG
			add_claim_by = VBG
		}
		ai_chance = { base = 60 }
	}
}

				}

		}
		ai_chance = { base = 65 }
	}
}

# South Caucasus: optional invitation to Caucasian Union
country_event = {
	id = dec.24
	title = dec.24.t
	desc = dec.24.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	option = {
		name = dec.24.a
		log = "[GetDateText]: [This.GetName]: dec.24.a accept Caucasian Union"
		ai_chance = {
			base = 55
			modifier = {
				factor = 1.5
				has_war_with = SOV
			}
			modifier = {
				factor = 0.4
				is_in_faction = yes
			}
		}
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		if = {
			limit = {
				FROM = {
					exists = yes
					is_faction_leader = yes
				}
			}
			set_country_flag = DEC_joined_named_faction
			FROM = { add_to_faction = ROOT }
		}
	}

	option = {
		name = dec.24.b
		log = "[GetDateText]: [This.GetName]: dec.24.b decline Caucasian Union"
		ai_chance = { base = 45 }
		add_political_power = 10
	}
}
